function getFiltered() {
  const q = document.getElementById('filter-search').value.toLowerCase().replace(' ','');
  const ft = document.getElementById('filter-type').value;
  const fsub = document.getElementById('filter-sub').value;
  const fg = document.getElementById('filter-grade').value;
  const fa = document.getElementById('filter-agency').value;
  const fs = document.getElementById('filter-status').value;
  return workers.filter(w=>{
    if(q&&!w.name.replace(/\s/g,'').toLowerCase().includes(q))return false;
    if(ft&&w.type!==ft)return false;
    if(fsub&&w.subtype!==fsub)return false;
    if(fg&&w.grade!==fg)return false;
    if(fa&&w.agency!==fa)return false;
    if(fs&&w.status!==fs)return false;
    if(filterDept&&getWorkerDept(w)!==filterDept)return false;
    return true;
  });
}

// ─── PDF 로컬 파싱 (API 없이) ────────────────────────────────
async function extractPdfText(file) {
  if(typeof pdfjsLib==='undefined') throw new Error('pdf.js 없음');
  pdfjsLib.GlobalWorkerOptions.workerSrc='https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
  const ab=await file.arrayBuffer();
  const pdf=await pdfjsLib.getDocument({data:ab}).promise;
  let text='';
  for(let i=1;i<=Math.min(pdf.numPages,3);i++){
    const page=await pdf.getPage(i);
    const tc=await page.getTextContent();
    text+=tc.items.map(it=>it.str).join(' ')+'\n';
  }
  return text;
}
function parseTextLocally(text){
  const r={};
  // 성명: (한글) 등 괄호 표기 제외하고 순수 이름만 추출
  // 성명 인식: "성명 한글 이소연" / "성명 이소연" 형식 모두 처리
  let nm = text.match(/성\s*명\s+한\s*글\s+([가-힣]{2,5})/);
  if(!nm) nm = text.match(/(?:성\s*명|이\s*름)\s+([가-힣]{2,5})(?!\s*(?:영문|한자|고등|대학))/);
  if(nm) r.name=nm[1];
  // 생년월일
  // 생년월일: 1994년 03월 25일 / 1994.03.25 / 1994-03-25 모두 처리
  let bd = text.match(/([0-9]{4})\s*년\s*([0-9]{1,2})\s*월\s*([0-9]{1,2})\s*일/);
  if(!bd) bd = text.match(/생년월일\s*[:\s]*([0-9]{4})[.\-\/]\s*([0-9]{1,2})[.\-\/]\s*([0-9]{1,2})/);
  if(bd) r.birth=`${bd[1]}-${String(bd[2]).padStart(2,'0')}-${String(bd[3]).padStart(2,'0')}`;
  // 연락처
  const ph=text.match(/(?:휴대폰|연락처|핸드폰)[\s:]*([0-9]{3}[-\s]?[0-9]{3,4}[-\s]?[0-9]{4})/);
  if(ph){const p=ph[1].replace(/\D/g,'');r.phone=`${p.slice(0,3)}-${p.slice(3,7)}-${p.slice(7)}`;}
  // 주소
  const ad=text.match(/(?:현\s*주\s*소|주\s*소)[\s:]*([가-힣\d\s\-,.()]+(?:동|로|길|구|시|도)[가-힣\d\s\-,.()]{2,60})/);
  if(ad) r.addr=ad[1].replace(/\s+/g,' ').trim().slice(0,80);
  // 파견업체
  const agMap=[
    ['KBS시큐리티',/KBS\s*시큐리티|kbssecurity/i],
    ['모닝인',/MORNING\s*IN|모닝인/i],
    ['프리머스HR',/PRIMUS\s*HR|프리머스/i],
    ['키스템프그룹',/KISTEMP|키스템프/i],

    ['피플씨앤에스',/피플씨앤에스|People\s*C.S/i],
    ['크릭앤리버',/CREEK.*RIVER|크릭앤리버/i],
    ['유니에스',/유니에스/i],
  ];
  for(const [nm2,re] of agMap){if(re.test(text)){r.agency=nm2;break;}}
  // 직군
  if(/인제스트|KDNS/.test(text)) r.type='인제스트';
  else if(/촬영보조|영상제작보조|카메라보조|오디오맨/.test(text)) r.type='촬영보조';

}

async function parseResume(file, sEl, sTxt, fp, lblPfx, resBoxId, resGridId, badgeId) {
  // 1차: Anthropic API (claude.ai 환경)
  try {
    setAiStatus(sEl, sTxt, 'loading', '이력서를 분석하고 있습니다...');
    const b64 = await new Promise((res,rej)=>{const r=new FileReader();r.onload=()=>res(r.result.split(',')[1]);r.onerror=rej;r.readAsDataURL(file);});
    const isPdf = file.type==='application/pdf';
    const ci = isPdf
      ? {type:'document',source:{type:'base64',media_type:'application/pdf',data:b64}}
      : {type:'image',source:{type:'base64',media_type:file.type,data:b64}};
    const resp = await fetch('https://api.anthropic.com/v1/messages',{
      method:'POST', headers:{'Content-Type':'application/json'},
      body:JSON.stringify({model:'claude-sonnet-4-20250514',max_tokens:800,messages:[{role:'user',content:[ci,{type:'text',text:RESUME_PROMPT}]}]})
    });
    if(!resp.ok) throw new Error('API '+resp.status);
    const data = await resp.json();
    if(data.error) throw new Error(data.error.message);
    const parsed = JSON.parse(data.content.map(c=>c.text||'').join('').replace(/```json|```/g,'').trim());
    const filled=[];
    if(parsed.name){document.getElementById(fp+'name').value=parsed.name;filled.push('name');}
    if(parsed.birth){document.getElementById(fp+'birth').value=parsed.birth;filled.push('birth');}
    if(parsed.phone){document.getElementById(fp+'phone').value=parsed.phone;filled.push('phone');}
    if(parsed.addr){const el=document.getElementById(fp+'addr');if(el){el.value=parsed.addr;filled.push('addr');}}

    if(parsed.type){const el=document.getElementById(fp+'type');if(el){el.value=parsed.type;onModalType(fp.replace(/-$/,''));filled.push('type');}}
    if(parsed.agency&&AGENCIES.includes(parsed.agency)){const el=document.getElementById(fp+'agency');if(el){el.value=parsed.agency;filled.push('agency');}}
    markFilled(filled, lblPfx);
    if(resBoxId) showAiResult(parsed, resBoxId, resGridId, badgeId);
    setAiStatus(sEl, sTxt, 'done', `AI 분석 완료 · ${filled.length}개 항목 추출`);
    return;
  } catch(e) { /* API 실패 → 로컬 파싱 */ }
  // 2차: PDF 로컬 파싱 (다운로드 버전)
  if(file.type==='application/pdf') {
    await parseResumeLocal(file, sEl, sTxt, fp, lblPfx, resBoxId, resGridId, badgeId);
  } else {
    setAiStatus(sEl, sTxt, 'err', '이미지는 PDF로 변환하거나 텍스트 붙여넣기를 이용해 주세요');
  }
}

  const w = workers.find(x=>x.id===id);

// 해당 월 실제 근무일수 & 일할 기본급 계산 (등급변경일 지원)
function calcProratedBase(w, year, month){
  const totalDays=getDaysInMonth(year,month);
  const monthStart=new Date(year,month-1,1);
  const monthEnd=new Date(year,month-1,totalDays);

  const cs=w.start?new Date(w.start):monthStart;
  const ce=w.end  ?new Date(w.end)  :monthEnd;
  const actualStart=cs>monthStart?cs:monthStart;
  const actualEnd  =ce<monthEnd  ?ce:monthEnd;
  if(actualStart>actualEnd) return {base:0,workDays:0,totalDays,hasGradeChange:false};
  const workDays=Math.round((actualEnd-actualStart)/86400000)+1;

  // 등급변경일이 해당 월에 있는지 확인
  const gcd=w.gradeChangeDate?new Date(w.gradeChangeDate):null;
  if(gcd && gcd>=actualStart && gcd<=actualEnd){
    // 변경 전 구간: actualStart ~ gcd-1 → 나등급
    const beforeDays=Math.round((gcd-actualStart)/86400000); // gcd 전날까지
    // 변경 후 구간: gcd ~ actualEnd → 가등급
    const afterDays=workDays-beforeDays;

    const naGrp=w.type==='촬영보조'?'shot':'ing_na';
    const gaGrp=w.type==='촬영보조'?'shot':'ing_ga';
    const naBase=getWage(year,naGrp).base;
    const gaBase=getWage(year,gaGrp).base;

    const baseBefore=Math.round(naBase/totalDays*beforeDays);
    const baseAfter =Math.round(gaBase/totalDays*afterDays);
    const combinedBase=baseBefore+baseAfter;

    return {base:combinedBase,workDays,totalDays,isProrated:true,
      hasGradeChange:true, gcdDay:gcd.getDate(),
      beforeDays,afterDays,baseBefore,baseAfter,
      naBase,gaBase,fullBase:gaBase};
  }

  const grp=getWageGroup(w);
  const fullBase=getWage(year,grp).base;
  // 결근일수 공제
  const absentKey=w.id+'_'+year+'_'+month;
  const absentDays=parseInt((payrollData[absentKey]||{})._absent||0)||0;
  const effectiveDays=Math.max(0,workDays-absentDays);
  const base=workDays===totalDays&&absentDays===0?fullBase:Math.round(fullBase/totalDays*effectiveDays);
  return {base,workDays,effectiveDays,absentDays,totalDays,
          isProrated:workDays!==totalDays||absentDays>0,hasGradeChange:false,fullBase};
}

function timeToMin(t){
  if(!t) return null;
  const [h,m]=(t+'').split(':').map(Number);
  return h*60+(m||0);
}

// 야간 구간(22:00~06:00)과의 겹치는 시간(분) 계산
function calcNightOverlap(wsMin, weMin){
  // 야간 구간: [0,360], [1320,1440], [1440,1800]
  const nights=[[0,360],[1320,1440],[1440,1800]];
  let total=0;
  nights.forEach(([ns,ne])=>{ total+=Math.max(0,Math.min(weMin,ne)-Math.max(wsMin,ns)); });
  return total;
}

// 4h 구간별 야간시간 정확 계산 (휴게가 야간에 발생하는 경우 반영)
function calcNightWorkSegmented(ws,we,brk){
  const BRK_PER=30,WORK_INTERVAL=240;
  let totalNight=0,pos=ws,brkLeft=brk;
  while(pos<we){
    const workEnd=Math.min(pos+WORK_INTERVAL,we);
    totalNight+=calcNightOverlap(pos,workEnd);
    pos=workEnd;
    if(brkLeft>0&&pos<we){pos+=BRK_PER;brkLeft-=BRK_PER;}
  }
  return totalNight;
}

// 소정근로(stdHours) 구간 내 야간시간 — OT야간 분리용
function calcNightInStd(ws,we,brk,stdHours){
  const BRK_PER=30,WORK_INTERVAL=240,stdMin=stdHours*60;
  let nightInStd=0,pos=ws,brkLeft=brk,workDone=0;
  while(pos<we&&workDone<stdMin){
    const segEnd=Math.min(pos+WORK_INTERVAL,we);
    const segLen=segEnd-pos;
    const stdPart=Math.min(segLen,stdMin-workDone);
    if(stdPart>0) nightInStd+=calcNightOverlap(pos,pos+stdPart);
    workDone+=segLen; pos=segEnd;
    if(brkLeft>0&&pos<we){pos+=BRK_PER;brkLeft-=BRK_PER;}
  }
  return nightInStd;
}

// 하루 근무 데이터로 시간외 분류 계산
function calcDayOT(dayData){
  const r={ot1:0,ot2:0,ot3:0,ot4:0,ot5:0,ot6:0,totalWork:0,stdWork:0,nightH:0};
  if(!dayData||!dayData.start||!dayData.end) return r;
  const ws=timeToMin(dayData.start);
  let we=timeToMin(dayData.end);
  if(ws===null||we===null) return r;
  // 익일 체크: nextDay 플래그 또는 퇴근 <= 출근이면 자동 익일
  if(dayData.nextDay || we<=ws) we+=1440;
  const rawMin=we-ws; // 휴게 전 총 경과시간(분)
  // 4시간마다 30분 자동 휴게: 4h미만=0, 4~8h=30분, 8h이상=60분
  const brk=Math.floor(rawMin/270)*30;
  const workMin=Math.max(0,rawMin-brk);
  const workH=workMin/60;
  r.totalWork=Math.round(workH*10)/10;
  const isHol=!!(dayData.holiday);
  const stdPerDay=8;

  // 야간 근무 시간 — 소정/OT 구간 정확 분리
  const nightTotalMin=calcNightWorkSegmented(ws,we,brk);
  const nightStdMin=calcNightInStd(ws,we,brk,stdPerDay);
  const nightOtMin=Math.max(0,nightTotalMin-nightStdMin);
  const nightMin=nightTotalMin; // 총 야간 (r.nightH용)
  const dayMin=Math.max(0,workMin-nightMin);
  const nightH=nightMin/60, dayH=dayMin/60;
  r.nightH=Math.round(nightH*10)/10; // 야간 별도

  if(!isHol){
    // 평일: 소정 8h, 초과 → 연장 (야간은 구간별 정확 분리)
    r.stdWork=Math.min(workH,stdPerDay);
    const totalOtH=Math.max(0,workH-stdPerDay);
    const nightInOt=nightOtMin/60;
    const dayOt=totalOtH-nightInOt;
    r.ot1=Math.round(Math.max(0,dayOt)*10)/10;    // 평일연장 ×1.5
    r.ot4=Math.round(Math.max(0,nightInOt)*10)/10; // 평일야간연장 ×2.0
  } else {
    // 휴일: 8h 이내→휴일근로, 초과→휴일연장 (야간 구간별 정확 분리)
    r.stdWork=0;
    const baseH=Math.min(workH,stdPerDay);
    const otH=Math.max(0,workH-stdPerDay);
    const nightInBase=nightStdMin/60;
    const nightInOt=nightOtMin/60;
    const dayBase=baseH-nightInBase;
    const dayOt=otH-nightInOt;
    r.ot2=Math.round(Math.max(0,dayBase)*10)/10;   // 휴일근로 ×1.5
    r.ot3=Math.round(Math.max(0,dayOt)*10)/10;     // 휴일연장 ×2.0
    r.ot5=Math.round(Math.max(0,nightInBase)*10)/10;// 휴일야간근로 ×2.0
    r.ot6=Math.round(Math.max(0,nightInOt)*10)/10;  // 휴일야간연장 ×2.5
  }
  return r;
}

// 월 합계 계산
function calcMonthPay(id,y,m){
  const w=workers.find(x=>x.id===id); if(!w) return {base:0,ot:0,total:0,totalH:0,stdH:0};
  const grp=getWageGroup(w);
  const wage=getWage(y,grp);
  const {base:proratedBase,workDays,totalDays:mDays,isProrated}=calcProratedBase(w,y,m);
  const data=getPayData(id,y,m);
  let ot=0,totalH=0,stdH=0;
  // 등급변경일 체크: 해당일 이전=나등급시급, 이후=가등급시급
  const _gcd=w.gradeChangeDate?new Date(w.gradeChangeDate):null;
  const _gcdMon=_gcd?_gcd.getFullYear()*100+_gcd.getMonth()+1:0;
  const _thisMon=y*100+m;
  const _hasGC=!!(_gcd && _gcdMon===_thisMon);
  const _gcdDay=_hasGC?_gcd.getDate():0;
  const _bhNa=billHourlyPreTax(getWage(y,w.type==='촬영보조'?'shot':'ing_na').base);
  const _bhGa=billHourlyPreTax(wage.base);
  Object.entries(data).forEach(([dayNum,day])=>{
    const dn=parseInt(dayNum);
    const dow=getDow(y,m,dn);
    const isHol=!!(day.holiday)||(dow===0||dow===6);
    const d=calcDayOT({...day,holiday:isHol});
    // 등급변경일 기준 시급 선택
    const bh=_hasGC&&dn<_gcdDay?_bhNa:_bhGa;
    const _h1=d.ot1+d.ot4, _h2=d.ot2+d.ot5, _h3=d.ot3+d.ot6, _hn=d.nightH||0;
    ot += (_h1*1.5 + _h2*1.5 + _h3*2.0 + _hn*0.5) * bh * 1.1;
    totalH+=d.totalWork;
    stdH+=d.stdWork;
  });
  const otH=Object.entries(data).reduce((s,[dayNum,day])=>{
    const dow=getDow(y,m,parseInt(dayNum));
    const isHol=!!(day.holiday)||(dow===0||dow===6);
    const d=calcDayOT({...day,holiday:isHol});
    return s+d.ot1+d.ot2+d.ot3+d.ot4+d.ot5+d.ot6;
  },0);
  return {base:proratedBase,fullBase:wage.base,ot:Math.round(ot),total:Math.round(proratedBase+ot),totalH:Math.round(totalH*10)/10,stdH:Math.round(stdH*10)/10,otH:Math.round(otH*10)/10,workDays,mDays,isProrated};
}

}

function buildDailyDetail(id,year,month){
  const w=workers.find(x=>x.id===id); if(!w) return '';
  const grp=getWageGroup(w);
  const wage=getWage(year,grp);
  const days=getDaysInMonth(year,month);
  const data=getPayData(id,year,month);
  const p=calcMonthPay(id,year,month);

  // 배율별 합계 누적
  let sumWork=0, sumOt4=[0,0,0,0], sumPay=0; // [평연,휴근,휴연,야간]
  const preCalc={};
  for(let d=1;d<=days;d++){
    const dd=data[d]||{};
    const dow=getDow(year,month,d);
    const isHol=!!(dd.holiday)||(dow===0||dow===6);
    const ot=calcDayOT({...dd,holiday:isHol});
    const _bhD=billHourlyPreTax(wage.base);
    const dayPay=Math.round(((ot.ot1+ot.ot4)*1.5+(ot.ot2+ot.ot5)*1.5+(ot.ot3+ot.ot6)*2.0+(ot.nightH||0)*0.5)*_bhD*1.1);
    sumWork+=ot.totalWork;
    sumOt4[0]=Math.round((sumOt4[0]+ot.ot1+ot.ot4)*10)/10;
    sumOt4[1]=Math.round((sumOt4[1]+ot.ot2+ot.ot5)*10)/10;
    sumOt4[2]=Math.round((sumOt4[2]+ot.ot3+ot.ot6)*10)/10;
    sumOt4[3]=Math.round((sumOt4[3]+(ot.nightH||0))*10)/10;
    sumPay+=dayPay;
    preCalc[d]={ot,dayPay};
  }
  const sumOtH=sumOt4[0]+sumOt4[1]+sumOt4[2]; // 야간 중복제외
  const KR=n=>Math.round(n).toLocaleString('ko-KR');

  // 타입·등급 배지
  const tb=w.type==='인제스트'
    ?`<span style="background:#E6F1FB;color:#185FA5;padding:3px 10px;border-radius:5px;font-size:12px;font-weight:500">인제스트</span>`
    :`<span style="background:#EEEDFE;color:#534AB7;padding:3px 10px;border-radius:5px;font-size:12px;font-weight:500">${w.subtype||'촬영보조'}</span>`;
  const gb=w.grade==='가'
    ?`<span style="background:#E6F1FB;color:#185FA5;padding:2px 8px;border-radius:5px;font-size:11px;font-weight:500">가등급</span>`
    :`<span style="background:#FAEEDA;color:#854F0B;padding:2px 8px;border-radius:5px;font-size:11px;font-weight:500">나등급</span>`;

  const overStd=sumWork>STD_HOURS;

  // 배율별 시간외 합계 뱃지 (합계 섹션용)
  const OT4_BG=['#E6F1FB','#E6F1FB','#EEEDFE','#1a1a2e'];
  const OT4_COL=['#185FA5','#185FA5','#534AB7','#a0a0e0'];
  const OT4_LBL=['평일연장','휴일근로','휴일연장','야간근로'];
  const OT4_RATE=[1.5,1.5,2.0,0.5];
  // 청구시급 = floor(기본급/262/10)*10, VAT포함 금액
  const _bh=billHourlyPreTax(wage.base); // VAT전 청구시급 (가:10700, 나:10400)
  const otSumBadges=sumOt4.map((h,i)=>h>0
    ?`<div style="background:${OT4_BG[i]};border-radius:8px;padding:8px 14px;min-width:120px">
        <div style="font-size:10px;color:${OT4_COL[i]};font-weight:600;margin-bottom:4px">${OT4_LBL[i]} ×${OT4_RATE[i]}×1.1</div>
        <div style="display:flex;align-items:baseline;gap:6px">
          <span style="font-size:15px;font-weight:700;color:${OT4_COL[i]}">${h.toFixed(1)}<span style="font-size:11px;font-weight:400">h</span></span>
          <span style="font-size:10px;color:${OT4_COL[i]};opacity:0.6">|</span>
          <span style="font-size:13px;font-weight:600;color:${OT4_COL[i]}">${KR(Math.round(_bh*OT4_RATE[i]*1.1*h))}<span style="font-size:10px;font-weight:400">원</span></span>
        </div>
      </div>`:''
  ).filter(Boolean).join('');

  // 날짜 행 생성
  let rows='';
  for(let d=1;d<=days;d++){
    const dow=getDow(year,month,d);
    const isWE=dow===0||dow===6;
    const dd=data[d]||{};
    const isHol=!!(dd.holiday)||isWE;
    const {ot,dayPay}=preCalc[d];
    const rawMinForBrk=(()=>{if(!dd.start||!dd.end)return 0;const ws2=timeToMin(dd.start);let we2=timeToMin(dd.end);if(we2<=ws2)we2+=1440;return we2-ws2;})();
    const brk=Math.floor(rawMinForBrk/270)*30;

    const isToday=new Date().getFullYear()===year&&new Date().getMonth()+1===month&&new Date().getDate()===d;
    const rowBg=isWE?'#F8F0FF':isHol&&!isWE?'#FFF8F0':isToday?'#F0F9FF':'#fff';

async function importAttendancePDF(input){
  const files = Array.from(input.files);
  if(!files.length) return;
  input.value = '';

  if(files.length === 1){
    // 단일 파일: 기존 방식
    await processPdfFile(files[0]);
    return;
  }

  // 여러 파일: 순차 처리 + 결과 누적
  const results = { ok:[], fail:[] };
  showAttResult('ok', `📄 PDF ${files.length}개 처리 중...`);

  for(let i=0; i<files.length; i++){
    const file = files[i];
    showAttResult('ok', `📄 처리 중 (${i+1}/${files.length}): ${file.name}`);
    try {
      const result = await processPdfFile(file, true); // silent=true
      if(result && result.ok) results.ok.push(result.msg);
      else results.fail.push(file.name + (result ? ': '+result.msg : ''));
    } catch(e) {
      results.fail.push(file.name + ': ' + e.message);
    }
  }

  lsSaveAll();
  // 최종 결과 표시
  let msg = `✅ ${results.ok.length}명 완료`;
  if(results.ok.length) msg += '\n' + results.ok.join('\n');
  if(results.fail.length) msg += `\n⚠ 실패 ${results.fail.length}건: ` + results.fail.join(', ');
  showAttResult(results.fail.length && !results.ok.length ? 'error' : 'ok', msg);
}

async function processPdfFile(file, silent=false){
  try {
    if(typeof pdfjsLib==='undefined') throw new Error('PDF.js 로드 실패');
    pdfjsLib.GlobalWorkerOptions.workerSrc='https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
    const ab = await file.arrayBuffer();
    const pdf = await pdfjsLib.getDocument({data:ab}).promise;

    // 모든 페이지 텍스트 아이템 수집
    let allItems = [];
    for(let pg=1; pg<=pdf.numPages; pg++){
      const page = await pdf.getPage(pg);
      const tc = await page.getTextContent();
      tc.items.forEach(it=>{
        if(it.str && it.str.trim())
          allItems.push({x:it.transform[4], y:it.transform[5], str:it.str});
      });
    }

    // Y좌표 기준 행 그룹핑 (허용오차 ±4)
    allItems.sort((a,b)=>b.y-a.y||a.x-b.x);
    const lineGroups = [];
    allItems.forEach(it=>{
      const last = lineGroups[lineGroups.length-1];
      if(last && Math.abs(it.y-last.baseY)<=4){
        last.items.push(it);
      } else {
        lineGroups.push({baseY:it.y, items:[it]});
      }
    });

    // 행별 X순 정렬 후 문자열 결합
    const textLines = lineGroups.map(g=>{
      const sorted = g.items.sort((a,b)=>a.x-b.x);
      let row = '';
      sorted.forEach((it,i)=>{
        if(i>0){
          const prev = sorted[i-1];
          const gap = it.x-(prev.x+prev.str.length*5.5);
          if(gap>8) row+=' ';
        }
        row+=it.str;
      });
      return row.trim();
    }).filter(Boolean);

    const fullText = textLines.join('\n');
    return parsePdfAttendanceText(fullText, silent);
  } catch(e) {
    if(!silent) showAttResult('error','PDF 읽기 실패: '+e.message);
    return {ok:false, msg:e.message};
  }
}

function parsePdfAttendanceText(text, silent=false){
  const lines = text.split('\n').map(l=>l.trim()).filter(Boolean);
  const flatText = lines.join(' ');

  // ── 연도/월 추출
  let year=0, month=0;
  const YM = /(\d{4})\s*년\s*(\d{1,2})\s*월/;
  for(const src of [flatText,...lines]){
    const m = src.match(YM);
    if(m){ year=parseInt(m[1]); month=parseInt(m[2]); break; }
  }
  if(!year||!month){
    const msg = 'PDF에서 연도/월을 찾을 수 없습니다.';
    if(!silent) showAttResult('error', msg);
    return {ok:false, msg};
  }

  // ── 성명 추출 (workers 목록과 직접 대조)
  let nameParsed = '';
  const NM_PATS = [
    /성\s*명\s*[\/\s부서]*\s*([가-힣]{2,5})/,
    /이\s*름\s*[:\s]*([가-힣]{2,5})/,
  ];
  outer: for(const src of [flatText,...lines]){
    for(const pat of NM_PATS){
      const m = src.match(pat);
      if(!m) continue;
      let cand = m[1];
      for(let len=5; len>=2; len--){
        const sub = cand.slice(0,len);
        if(workers.find(w=>w.name.trim().replace(/\s/g,'')===sub)){
          nameParsed=sub; break outer;
        }
      }
      if(!nameParsed) nameParsed=cand.slice(0,4);
    }
    if(nameParsed) break;
  }
  // fallback: workers 이름을 텍스트에서 직접 검색
  if(!nameParsed){
    for(const w of workers){
      const nm = w.name.trim().replace(/\s/g,'');
      if(nm.length>=2 && flatText.includes(nm)){ nameParsed=nm; break; }
    }
  }
  if(!nameParsed){
    const msg = 'PDF에서 성명을 찾을 수 없습니다.';
    if(!silent) showAttResult('error', msg);
    return {ok:false, msg};
  }

  const worker = workers.find(w=>w.name.trim().replace(/\s/g,'')===nameParsed.replace(/\s/g,''));
  if(!worker){
    const msg = `"${nameParsed}" 직원을 목록에서 찾을 수 없습니다.`;
    if(!silent) showAttResult('error', msg);
    return {ok:false, msg};
  }

  // ── 날짜 행 파싱 (2단계: 날짜 → 시간)
  const DATE_SIMPLE = /(\d{4})\s*[.\s]\s*(\d{1,2})\s*[.\s]\s*(\d{1,2})\s*(일|월|화|수|목|금|토)\s*(휴일)?/;
  const TIME_FIND   = /\b([01]?\d|2[0-3])\s*:\s*([0-5]\d)\b/g;
  const data = getPayData(worker.id, year, month);
  let imported=0, skipped=0;

  for(const line of lines){
    const dm = DATE_SIMPLE.exec(line);
    if(!dm) continue;
    const y2=parseInt(dm[1]), mo=parseInt(dm[2]), d=parseInt(dm[3]);
    if(y2!==year || mo!==month) continue;
    const holKW = dm[5];

    // 날짜 패턴 이후 텍스트에서만 시간 검색 (잘못된 숫자 오인식 방지)
    const lineAfterDate=line.slice(dm.index+dm[0].length);
    TIME_FIND.lastIndex=0;
    const times=[];
    let tm;
    while((tm=TIME_FIND.exec(lineAfterDate))!==null){
      times.push(String(parseInt(tm[1])).padStart(2,'0')+':'+tm[2]);
    }
    const start = times[0]||'';
    const end   = times[1]||'';
    if(!start){ skipped++; continue; }

    const toMin = t=>parseInt(t.split(':')[0])*60+parseInt(t.split(':')[1]);
    const nextDay = !!(start&&end&&toMin(end)<=toMin(start));
    const isWE = new Date(year,month-1,d).getDay();
    const isHol = !!holKW && !(isWE===0||isWE===6);

    if(!data[d]) data[d]={};
    data[d].start=start; data[d].end=end;
    if(nextDay) data[d].nextDay=true; else delete data[d].nextDay;
    data[d].holiday=isHol;
    imported++;
  }

  // 연도/월 셀렉트 동기화 (단일 파일 모드일 때만)
  if(!silent){
    const ySel=document.getElementById('pay-year');
    const mSel=document.getElementById('pay-month');
    if(ySel){
      let found=false;
      for(const o of ySel.options){if(parseInt(o.value)===year){ySel.value=o.value;found=true;break;}}
      if(!found){const o=document.createElement('option');o.value=year;o.textContent=year+'년';ySel.appendChild(o);ySel.value=year;}
    }
    if(mSel){for(const o of mSel.options){if(parseInt(o.value)===month){mSel.value=o.value;break;}}}
    lsSaveAll();
    openDetail(worker.id);
  }

  const msg = `${worker.name.trim()} ${year}년 ${month}월 — ${imported}일`+(skipped?` (비근무 ${skipped}일 제외)`:'');
  if(!silent) showAttResult('ok','✅ '+msg);
  return {ok:true, msg};
}

function importAttendance(input){
  const file=input.files[0]; if(!file) return;
  input.value='';
  const reader=new FileReader();
  reader.onload=function(ev){
    try{
      const wb=XLSX.read(ev.target.result,{type:'array',cellDates:false,raw:true});
      const ws=wb.Sheets[wb.SheetNames[0]];
      if(!ws){showAttResult('error','시트를 읽을 수 없습니다');return;}

      // ── 셀 값 추출 헬퍼 ──────────────────────────────────
      const cv = addr => ws[addr] || null;

      // 숫자 반환 (에러셀/빈셀 → null, 문자열 → 숫자변환 시도)
      const gN = addr => {
        const c=cv(addr); if(!c) return null;
        if(c.t==='e') return null;
        if(c.t==='n') return (c.v===null||c.v===undefined) ? null : c.v;
        const n=parseFloat(c.v); return isNaN(n)?null:n;
      };

      // 문자열 반환 (에러셀/빈셀 → '')
      const gS = addr => {
        const c=cv(addr); if(!c||c.t==='e') return '';
        const v=c.v; if(v===null||v===undefined) return '';
        return String(v).trim();
      };

      // HHMM 정수 → "HH:MM" (예: 1400 → "14:00")
      const hhmm = n => {
        if(n===null||n===undefined) return '';
        const v=Math.round(Number(n));
        if(isNaN(v)||v<100||v>2359) return '';
        const h=Math.floor(v/100), m=v%100;
        if(h>23||m>59) return '';
        return `${String(h).padStart(2,'0')}:${String(m).padStart(2,'0')}`;
      };

      // 엑셀 날짜 시리얼 → {y,m,d}
      const sd2date = s => {
        if(!s||s<40000) return null;
        const d=new Date(Math.round((s-25569)*86400*1000));
        return isNaN(d.getTime())?null:{y:d.getUTCFullYear(),m:d.getUTCMonth()+1,d:d.getUTCDate()};
      };

      // ── 연도·월 읽기 ──────────────────────────────────────
      const year  = parseInt(gN('P4')||gS('P4'))||0;
      const month = parseInt(gN('P5')||gS('P5'))||0;
      if(!year||!month){showAttResult('error',`연도/월을 찾을 수 없습니다\nP4=${gS('P4')} P5=${gS('P5')}`);return;}

      // ── 이름 읽기 (D5 우선, C5 대안) ─────────────────────
      const nameRaw = gS('D5')||gS('C5');
      const nameParsed = nameRaw.includes('/')? nameRaw.split('/').pop().trim() : nameRaw;
      if(!nameParsed){showAttResult('error',`성명을 찾을 수 없습니다 (D5="${gS('D5')}")`);return;}

      const worker = workers.find(w=>
        w.name.trim().replace(/\s/g,'') === nameParsed.replace(/\s/g,'')
      );
      if(!worker){showAttResult('error',`"${nameParsed}" — 직원 목록에 없습니다`);return;}

      const data = getPayData(worker.id, year, month);
      let imported=0, skipped=0;

      // ── 날짜 행 루프 ──────────────────────────────────────
      for(let row=10; row<=45; row++){
        // 날짜 시리얼 (B열 우선, C열 대안)
        const bN=gN('B'+row), cN=gN('C'+row);
        const serial=(bN&&bN>40000)?bN:(cN&&cN>40000)?cN:null;
        if(!serial) continue;
        const dt=sd2date(serial);
        if(!dt||dt.y!==year||dt.m!==month) continue;

        // 출근시간: P열(출근시각) 우선, D열(시업시간) 대안
        const pN=gN('P'+row), dN=gN('D'+row);
        if(pN===0){skipped++;continue;}          // P=0 → 결근/휴가
        const start = hhmm(pN) || hhmm(dN);
        if(!start){skipped++;continue;}           // 근무 없음

        // 퇴근시간: E열
        const eC = cv('E'+row);
        let end='', nextDay=false;
        if(eC && eC.v!==null && eC.v!==undefined){
          if(eC.t==='n' && Number(eC.v)>=100){
            end = hhmm(Number(eC.v));
          } else {
            const es = String(eC.v);
            const hasNext = es.includes('익일')||es.includes('D+')||es.includes('d+');
            const tm = es.match(/(\d{1,2}):(\d{2})/);
            if(tm){
              end = String(parseInt(tm[1])).padStart(2,'0')+':'+tm[2];
              if(hasNext) nextDay=true;
            } else {
              const en=parseFloat(es);
              if(!isNaN(en)&&en>=100) end=hhmm(en);
            }
          }
        }

        // 휴일 여부 (O열)
        const oStr = gS('O'+row);
        const isHol = oStr.includes('휴');
        const dow = new Date(year,month-1,dt.d).getDay();

        // 저장
        if(!data[dt.d]) data[dt.d]={};
        data[dt.d].start   = start;
        data[dt.d].end     = end||'';
        data[dt.d].holiday = isHol&&dow!==0&&dow!==6;
        if(nextDay) data[dt.d].nextDay=true;
        else        delete data[dt.d].nextDay;
        imported++;
      }

      // 연도/월 셀렉트 동기화
      const ySel=document.getElementById('pay-year');
      const mSel=document.getElementById('pay-month');
      if(ySel){
        let f=false;
        for(const o of ySel.options){if(parseInt(o.value)===year){ySel.value=o.value;f=true;break;}}
        if(!f){const o=document.createElement('option');o.value=year;o.textContent=year+'년';ySel.appendChild(o);ySel.value=year;}
      }
      if(mSel) for(const o of mSel.options){if(parseInt(o.value)===month){mSel.value=o.value;break;}}

      lsSaveAll();
      openDetail(worker.id);
      showAttResult('ok',
        `✅ ${nameParsed} ${year}년 ${month}월 — ${imported}일 업로드 완료`+
        (skipped?` (결근/휴가 ${skipped}일 제외)`:'')
      );
    }catch(err){
      showAttResult('error','파일 읽기 실패: '+err.message);
      console.error('[importAttendance]',err);
    }
  };
  reader.readAsArrayBuffer(file);
}

function autoFmtTime(input){
  let v=input.value.replace(/[^0-9:]/g,'');
  // 4자리 숫자만 HH:MM으로 자동변환 (3자리 규칙 제거 — 110 → 01:10 오저장 버그 방지)
  if(/^\d{4}$/.test(v)) v=v.slice(0,2)+':'+v.slice(2);
  input.value=v;
  // HH:MM 완성 시 즉시 저장
  if(/^\d{2}:\d{2}$/.test(v)){
    input.dispatchEvent(new Event('blur'));
  }
}

// 해당 월 청구 대상 여부 (근무중 + 계약종료 다음달까지 포함)
function isWorkerBillableInMonth(w, year, month){
  const monthEnd=new Date(year,month,0);
  const prevMonthStart=new Date(year,month-2,1);
  // ① 시작일이 이번 달 이후면 제외
  if(w.start && new Date(w.start)>monthEnd) return false;
  // ② 근무중이면 포함
  if(w.status==='근무중') return true;
  // ③ 계약종료: 종료일이 지난달 이후면 포함 (이번달 종료 → 다음달 청구 포함)
  if(w.end) return new Date(w.end)>=prevMonthStart;
  return false;
}
function buildInvoiceData(year, month, agency){
  const DEPTS=['영상취재1부','영상취재2부','뉴스영상콘텐츠부'];
  const result={};
  DEPTS.forEach(d=>result[d]=[]);
  workers.forEach(w=>{
    if(!isWorkerBillableInMonth(w,year,month)) return;
    if(agency && w.agency!==agency) return;
    const dept=getWorkerDept(w);
    if(!result[dept]) result[dept]=[];
    const p=calcMonthPay(w.id,year,month);
    result[dept].push({w,p});
  });
  return result;
}

function downloadInvoiceExcel(){
  const year=parseInt(document.getElementById('inv-year')?.value)||new Date().getFullYear();
  const month=parseInt(document.getElementById('inv-month')?.value)||new Date().getMonth()+1;
  const agency=document.getElementById('inv-agency')?.value||'';
  const mDays=getDaysInMonth(year,month);

  // ── 상수 ──────────────────────────────────────────────────
  const INDIRECT = 0.117263;  // 간접비율 (4대보험 사용자+관리비)

  // ── 부서 그룹 ──────────────────────────────────────────────
  const GROUPS=[
    {label:'영상취재1부 데일리',     filter:w=>w.type==='촬영보조'&&w.subtype==='데일리'},
    {label:'영상취재2부 특집/스포츠',filter:w=>w.type==='촬영보조'&&(w.subtype==='특집'||w.subtype==='스포츠')},
    {label:'뉴스영상콘텐츠부',      filter:w=>w.type==='인제스트'},
  ];

  const aoa=[];   // 전체 rows
  const merges=[]; // merged cell ranges
  let r=0;        // 현재 행 (0-based)

  function addRow(arr){ aoa.push(arr); r++; }
  function blank(){ aoa.push([]); r++; }
  function mg(r1,c1,r2,c2){ merges.push({s:{r:r1,c:c1},e:{r:r2,c:c2}}); }

  GROUPS.forEach(group=>{
    const targets=workers.filter(w=>
      isWorkerBillableInMonth(w,year,month)&&group.filter(w)&&(!agency||w.agency===agency)
    );
    if(!targets.length) return;

    // ── 제목 행 ──────────────────────────────────────────────
    const titleRow=new Array(24).fill('');
    titleRow[0]=`KBS 보도영상 ${group.label} ${year}년 ${month}월 청구내역서`;
    addRow(titleRow); mg(r-1,0,r-1,23);

    const agencyRow=new Array(24).fill('');
    agencyRow[0]=`업체명: ${agency||'전체'}`;
    addRow(agencyRow); mg(r-1,0,r-1,23);

    // ── 헤더 3행 ────────────────────────────────────────────
    const hR=r; // 헤더 시작 행
    const h0=new Array(24).fill('');
    h0[0]='순번'; h0[1]='성명'; h0[2]='근무일수'; h0[3]='용역비';
    h0[4]='결근일수'; h0[5]='결근공제';
    h0[6]='대휴 및 야근 공제\n(VAT포함)';
    h0[8]='용역비\n소 계\n(VAT포함)';
    h0[9]='시간외수당';
    h0[17]='소  계\n(VAT포함)';
    h0[18]='시간외수당\n지급기준';
    h0[19]='시간외수당\n실지급액';
    h0[20]='시간외수당\n간접비';
    h0[21]='퇴직급여\n충당금';
    h0[22]='합   계\n(VAT포함)';
    h0[23]='비   고\n(시간합계)';
    addRow(h0);

    const h1=new Array(24).fill('');
    h1[6]='일수'; h1[7]='금액';
    h1[9]='평일연장근로수당\n[시급×1.5×1.1]';
    h1[11]='휴일근로수당\n[시급×1.5×1.1]';
    h1[13]='휴일연장근로수당\n[시급×2.0×1.1]';
    h1[15]='야간근로수당\n[시급×0.5×1.1]';
    addRow(h1);

    const h2=new Array(24).fill('');
    h2[9]='시간'; h2[10]='금액';
    h2[11]='시간'; h2[12]='금액';
    h2[13]='시간'; h2[14]='금액';
    h2[15]='시간'; h2[16]='금액';
    addRow(h2);

    // 헤더 병합
    [0,1,2,3,4,5,8,17,18,19,20,21,22,23].forEach(c=>mg(hR,c,hR+2,c)); // 3행 통합
    mg(hR,6,hR,7);        // 대휴G+H 가로병합(1행)
    mg(hR,9,hR,16);       // 시간외수당 J~Q 가로병합
    mg(hR+1,9,hR+1,10);   // 평일연장 J+K
    mg(hR+1,11,hR+1,12);  // 휴일근로 L+M
    mg(hR+1,13,hR+1,14);  // 휴일연장 N+O
    mg(hR+1,15,hR+1,16);  // 야간 P+Q
    [17,18,19,20,21,22,23].forEach(c=>mg(hR,c,hR+1,c)); // R~X 2행 병합

    // ── 데이터 행 ────────────────────────────────────────────
    let sB=0,sAbs=0,sDeduct=0,sI=0,sJ=0,sK=0,sL=0,sM=0,sN=0,sO=0,sP=0,sQ=0;
    let sR=0,sT=0,sU=0,sV=0,sW=0,sX=0;

    targets.forEach((w,i)=>{
      const p=calcMonthPay(w.id,year,month);
      const fullBase=getWage(year,getWageGroup(w)).base;
      const workDays=p.workDays||mDays;
      const absentN=parseInt((payrollData[`${w.id}_${year}_${month}`]||{})._absent||0)||0;
      const D_val=workDays===mDays?fullBase:Math.round(fullBase/mDays*workDays);
      const F_val=absentN>0?Math.round(fullBase/mDays*absentN):0;
      const I_val=p.base;
      const bh=billHourlyPreTax(fullBase);

      // 수당 계산
      const data=getPayData(w.id,year,month);
      let h1h=0,h2h=0,h3h=0,nh=0;
      Object.entries(data).forEach(([dn,day])=>{
        if(isNaN(parseInt(dn)))return;
        const dow=getDow(year,month,parseInt(dn));
        const isHol=!!(day.holiday)||(dow===0||dow===6);
        const ot=calcDayOT({...day,holiday:isHol});
        h1h+=ot.ot1+ot.ot4;
        h2h+=ot.ot2+ot.ot5;
        h3h+=ot.ot3+ot.ot6;
        nh+=ot.nightH||0;
      });
      h1h=Math.round(h1h*10)/10;
      h2h=Math.round(h2h*10)/10;
      h3h=Math.round(h3h*10)/10;
      nh=Math.round(nh*10)/10;

      const K=Math.round(h1h*bh*1.5*1.1);
      const M=Math.round(h2h*bh*1.5*1.1);
      const O=Math.round(h3h*bh*2.0*1.1);
      const Q=Math.round(nh*bh*0.5*1.1);
      const R=K+M+O+Q;
      const U=Math.round(R*INDIRECT);
      const V=Math.round(R/12*10)/10;
      const W=Math.round(I_val+R+U+V);
      const X=Math.round((h1h+h2h+h3h+nh)*10)/10;

      addRow([
        i+1, w.name.trim(), workDays, D_val,
        absentN, F_val, 0, 0, I_val,
        h1h, K, h2h, M, h3h, O, nh, Q,
        R, 0, R, U, V, W, X
      ]);

      sB+=D_val; sAbs+=absentN; sDeduct+=F_val; sI+=I_val;
      sJ+=h1h;sK+=K;sL+=h2h;sM+=M;sN+=h3h;sO+=O;sP+=nh;sQ+=Q;
      sR+=R;sT+=R;sU+=U;sV+=V;sW+=W;sX+=X;
    });

    // 합계 행
    addRow(['','합계','',sB,sAbs,sDeduct,0,0,sI,
      Math.round(sJ*10)/10,sK,Math.round(sL*10)/10,sM,
      Math.round(sN*10)/10,sO,Math.round(sP*10)/10,sQ,
      sR,0,sT,sU,Math.round(sV*10)/10,Math.round(sW),Math.round(sX*10)/10]);

    blank();

    // 공급가액/부가세/합계 행
    const totW=Math.round(sW);
    const supply=Math.round(totW/1.1);
    const vat=totW-supply;
    const vatRow=new Array(24).fill('');
    vatRow[0]='공급가액'; vatRow[3]=supply;
    vatRow[8]='부가세'; vatRow[9]=vat;
    vatRow[11]='합  계'; vatRow[15]=totW;
    addRow(vatRow);

    blank(); blank();
  });

  // 시트 생성
  const ws=XLSX.utils.aoa_to_sheet(aoa);
  ws['!merges']=merges;
  ws['!cols']=[
    {wch:6},{wch:10},{wch:8},{wch:14},{wch:8},{wch:14},
    {wch:6},{wch:14},{wch:14},{wch:6},{wch:14},
    {wch:6},{wch:14},{wch:6},{wch:14},{wch:6},{wch:14},
    {wch:14},{wch:8},{wch:14},{wch:14},{wch:14},{wch:16},{wch:10}
  ];

  const wb=XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb,ws,`${month}월 청구내역서`);
  XLSX.writeFile(wb,`${year}년${month}월_청구내역서${agency?'_'+agency:''}.xlsx`);
  document.getElementById('modal-invoice').style.display='none';
}

function renderRank(){
  const year=parseInt(document.getElementById('rank-year')?.value)||new Date().getFullYear();
  const month=parseInt(document.getElementById('rank-month')?.value)||new Date().getMonth()+1;
  const ftype=document.getElementById('rank-type')?.value||'';
  const fdept=document.getElementById('rank-dept')?.value||'';

  const targets=workers.filter(w=>{
    if(!isWorkerBillableInMonth(w,year,month)) return false;
    if(ftype&&w.type!==ftype) return false;
    if(fdept&&getWorkerDept(w)!==fdept) return false;
    return true;
  });

  const KR=n=>Math.round(n).toLocaleString('ko-KR');

  const rows=targets.map(w=>{
    const p=calcMonthPay(w.id,year,month);
    const data=getPayData(w.id,year,month);
    let h1=0,h2=0,h3=0,nh=0;
    Object.entries(data).forEach(([dn,day])=>{
      if(isNaN(parseInt(dn)))return;
      const dow=getDow(year,month,parseInt(dn));
      const isHol=!!(day.holiday)||(dow===0||dow===6);
      const ot=calcDayOT({...day,holiday:isHol});
      h1+=ot.ot1+ot.ot4; h2+=ot.ot2+ot.ot5;
      h3+=ot.ot3+ot.ot6; nh+=ot.nightH||0;
    });
    h1=Math.round(h1*10)/10; h2=Math.round(h2*10)/10;
    h3=Math.round(h3*10)/10; nh=Math.round(nh*10)/10;
    const totalOT=Math.round((h1+h2+h3+nh)*10)/10;
    return {w,p,h1,h2,h3,nh,totalOT};
  }).sort((a,b)=>b.totalOT-a.totalOT);

  // 요약 카드
  const worked=rows.filter(r=>r.totalOT>0);
  const maxOT=rows[0]?.totalOT||0;
  const avgOT=worked.length?Math.round(worked.reduce((s,r)=>s+r.totalOT,0)/worked.length*10)/10:0;
  const totalOTSum=Math.round(rows.reduce((s,r)=>s+r.totalOT,0)*10)/10;
  const totalPay=rows.reduce((s,r)=>s+r.p.ot,0);

  document.getElementById('rank-summary').innerHTML=`

    <div class="scard"><div class="lbl">1인 평균</div><div class="val val-green">${avgOT}<span style="font-size:14px;font-weight:400"> h</span></div></div>
    <div class="scard" style="background:#FFF3E8;border-color:#FAEEDA"><div class="lbl" style="color:#854F0B">시간외수당 합계</div><div class="val val-amber">${KR(totalPay)}<span style="font-size:13px;font-weight:400"> 원</span></div></div>`;

  if(!rows.length){
    document.getElementById('rank-content').innerHTML='<div style="text-align:center;padding:3rem;color:#aaa">해당 월 데이터가 없습니다.</div>';
    return;
  }

  const MEDAL=['🥇','🥈','🥉'];
  let html=`<div class="tbl-wrap"><table style="min-width:900px">
    <thead><tr>
      <th style="width:48px;text-align:center">순위</th>
      <th>성명 / 업체</th>
      <th>부서</th>
      <th>직군</th>
      <th style="text-align:right;color:#185FA5">평일연장</th>
      <th style="text-align:right;color:#185FA5">휴일근로</th>
      <th style="text-align:right;color:#534AB7">휴일연장</th>
      <th style="text-align:right;color:#2d6a2d">야간</th>
      <th style="text-align:right;color:#1a1a1a;background:#EBF4FF">합계 (h)</th>
      <th style="text-align:right;color:#854F0B">시간외수당</th>
      <th style="width:120px">비율</th>
    </tr></thead>
    <tbody>`;

  rows.forEach((r,i)=>{
    const {w,p,h1,h2,h3,nh,totalOT}=r;
    const medal=i<3?MEDAL[i]:`<span style="color:#aaa;font-size:12px">${i+1}</span>`;
    const bar=maxOT>0?Math.round(totalOT/maxOT*100):0;
    const isTop=i<3;
    const isEnded=w.status==='계약종료';
    const bg=i===0?'#FFFDE7':i===1?'#F1F8E9':i===2?'#FFF3E8':'';
    html+=`<tr style="background:${bg}">
      <td style="text-align:center;font-size:${isTop?'18':'12'}px;padding:8px 6px">${medal}</td>
      <td style="padding:8px 10px">
        <span style="font-weight:600">${w.name.trim()}</span>
        ${isEnded?'<span style="background:#FCEBEB;color:#A32D2D;font-size:9px;padding:1px 5px;border-radius:3px;margin-left:4px">종료</span>':''}

// ── 1. 서버(구글 드라이브)에 데이터 저장하기 ────────────
async function saveDataToSheet(data) {
  const url = getGasUrl();
  if (!url) { openGasSetup(); return; }
  setCloudStatus('저장 중...', '#185FA5');
  try {
    // form-encoded POST: GAS 리다이렉트 후에도 e.parameter.d 로 데이터 수신
    const body = 'action=save&d=' + encodeURIComponent(JSON.stringify(data));
    const response = await fetch(url, {
      method: 'POST',
      headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
      body
    });
    const result = await response.json();
    if (result.ok) {
      window._myLastSaveTime = Date.now();
      setCloudStatus('✅ 서버에 저장됐습니다.', '#0F6E56');
      showSaveToast('서버 저장 완료', 'save');
    } else {
      throw new Error(result.msg || '저장 실패');
    }
  } catch (error) {
    console.error('저장 실패:', error);
    const msg = error.message.includes('fetch')
      ? 'GAS URL 확인 또는 🔍 테스트 버튼으로 연결 확인 후 다시 시도해 주세요.'
      : error.message;
    setCloudStatus('업로드 실패: ' + msg, '#A32D2D');
    showSaveToast('업로드 실패', 'warn');
  }
}

// ── 2. 서버(구글 드라이브)에서 데이터 불러오기 ──────────
async function loadDataFromSheet() {
  const url = getGasUrl();
  if (!url) { openGasSetup(); return; }
  setCloudStatus('불러오는 중...', '#185FA5');
  try {
    const response = await fetch(url, { method: 'GET' });
    const result = await response.json();
    if (!result.ok) throw new Error(result.msg || '실패');
    const data = result.data;
    if (!data || !data.workers) {
      setCloudStatus('저장된 데이터가 없습니다. 관리자가 먼저 저장해야 합니다.', '#854F0B');
      return;
    }
    if (!confirm(`서버 데이터(직원 ${data.workers?.length||0}명)를 가져올까요?\n현재 데이터가 교체됩니다.`)) return;
    // 가져온 데이터를 프로그램 변수에 할당하고 화면 갱신
    if (data.workers && Array.isArray(data.workers) && data.workers.length > 0) {
      workers = data.workers;
      nextId = Math.max(...workers.map(x => x.id), 399) + 1;
    }
    if (data.history)      history = data.history;
    if (data.payrollData)  payrollData = data.payrollData;
    if (data.wageSettings) Object.assign(wageSettings, data.wageSettings);
    if (data.extraRates)   Object.assign(extraRates, data.extraRates);
    if (data.agencyContacts) Object.assign(agencyContacts, data.agencyContacts || {});
    lsSaveAll();
    renderSummary();
    renderTable();   // 화면 갱신 (renderStaffList 역할)
    setCloudStatus(`✅ 서버 데이터를 성공적으로 불러왔습니다. (직원 ${workers.length}명)`, '#0F6E56');
    showSaveToast('서버 데이터 불러오기 완료', 'save');
    startGasAutoSync();
  } catch (error) {
    console.error('불러오기 실패:', error);
    setCloudStatus('불러오기 실패: ' + error.message, '#A32D2D');
  }
}

// ── 자동 동기화 (60초마다) ────────────────────────────
function startGasAutoSync() {
  if (gasAutoInterval) clearInterval(gasAutoInterval);
  const url = getGasUrl();
  if (!url) return;
  const ind = document.getElementById('sync-indicator');
  if (ind) ind.style.display = 'inline-flex';
  gasAutoInterval = setInterval(async () => {
    try {
      const res = await fetch(url, { method: 'GET' });
      const result = await res.json();
      if (!result.ok || !result.data || !result.data.workers) return;
      const data = result.data;
      const remoteTs = data.savedAt ? new Date(data.savedAt).getTime() : 0;
      const localTs  = parseInt(localStorage.getItem('kbs_last_sync_ts') || 0);
      const myLast   = window._myLastSaveTime || 0;
      if (remoteTs > localTs && Date.now() - myLast > 3000) {
        workers = data.workers;
        nextId = Math.max(...workers.map(x => x.id), 399) + 1;
        if (data.history)      history = data.history;
        if (data.payrollData)  payrollData = data.payrollData;
        if (data.agencyContacts) Object.assign(agencyContacts, data.agencyContacts || {});
        lsSaveAll();
        renderSummary();
        renderTable();
        localStorage.setItem('kbs_last_sync_ts', remoteTs);
        const dt = new Date(remoteTs).toLocaleTimeString('ko-KR', { hour: '2-digit', minute: '2-digit' });
        setCloudStatus(`🔄 자동 동기화됨 — ${dt}`, '#185FA5');
      }
    } catch (e) { /* 무시 */ }
  }, 60000);
}

async function testGasUrl() {

    const res    = await fetch(url, { method: 'GET' });
    const result = await res.json();
    if (!result.ok && result.ok !== undefined) throw new Error(result.msg || '서버 오류');
    if (el) { el.textContent = '✅ 연결 성공! 데이터: ' + (result.data ? '있음' : '없음(첫 업로드 전)'); el.style.color = '#0F6E56'; }
