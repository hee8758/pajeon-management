function doPost(e) {
  try {
    let data;
    // form-encoded 방식: e.parameter.d
    if (e.parameter && e.parameter.d) {
      data = JSON.parse(e.parameter.d);
    }
    // text/plain 방식: e.postData.contents
    else if (e.postData && e.postData.contents) {
      const body = JSON.parse(e.postData.contents);
      data = body.data || body;
    }
    if (!data) throw new Error('데이터 없음');
    getDataFile().setContent(JSON.stringify(data));
    return ContentService
      .createTextOutput(JSON.stringify({ ok: true }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, msg: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    const files = DriveApp.getFilesByName(FILE_NAME);
    if (!files.hasNext()) return ContentService

      .setMimeType(ContentService.MimeType.JSON);
    const raw = files.next().getBlob().getDataAsString();
    return ContentService

  } catch (err) {
    return ContentService
